# Shadows of the Blood - Blood Feud Casus Belli Engine

sotb_blood_feud_war = {
	name = CB_NAME_SOTB_BLOOD_FEUD
	war_name = WAR_NAME_SOTB_BLOOD_FEUD
	sprite = 1
	truce_days = 3650
	hostile_against_others = yes
	is_permanent = yes
	check_all_titles = yes
	press_claim = yes
	can_ask_to_join_war = yes
	infamy_modifier = 0.5 # 50% Infamy reduction for blood vendetta
	
	can_use_gui = {
		ROOT = {
			has_trait = sotb_blood_feudist
			is_ruler = yes
			is_adult = yes
			prisoner = no
			is_incapable = no
		}
		FROM = {
			is_ruler = yes
			OR = {
				has_trait = sotb_blood_feudist
				is_rival = ROOT
				has_opinion_modifier = { who = ROOT modifier = opinion_sotb_blood_feud }
				reverse_has_opinion_modifier = { who = ROOT modifier = opinion_sotb_blood_feud }
			}
		}
	}
	
	can_use = {
		ROOT = {
			has_trait = sotb_blood_feudist
			is_ruler = yes
			is_adult = yes
			prisoner = no
			is_incapable = no
		}
		FROM = {
			is_ruler = yes
			OR = {
				has_trait = sotb_blood_feudist
				is_rival = ROOT
				has_opinion_modifier = { who = ROOT modifier = opinion_sotb_blood_feud }
				reverse_has_opinion_modifier = { who = ROOT modifier = opinion_sotb_blood_feud }
			}
		}
	}
	
	on_success = {
		# Attacker Victorious
		ROOT = {
			prestige = 250
			add_character_modifier = {
				name = sotb_feud_victor
				duration = 3650
			}
			character_event = { id = sotb_feud.25 }
		}
		
		# Wergild Reparations Transfer
		FROM = {
			transfer_scaled_wealth = {
				to = ROOT
				value = 1.5
				min = 150
				max = 600
			}
			prestige = -250
			add_character_modifier = {
				name = sotb_feud_vanquished
				duration = 3650
			}
		}
		
		# Title transfer: Usurp targeted disputed/ancestral border county if within de jure or claim
		any_attacker = {
			limit = { character = ROOT }
			participation_scaled_prestige = 100
		}
	}
	
	on_fail = {
		# White Peace
		ROOT = {
			prestige = -100
		}
		FROM = {
			prestige = 100
		}
	}
	
	on_reverse_demand = {
		# Defender Crushes Attacker
		ROOT = {
			prestige = -400
			add_character_modifier = {
				name = sotb_feud_vanquished
				duration = 5475 # 15 years humiliation
			}
			character_event = { id = sotb_feud.26 }
		}
		
		FROM = {
			transfer_scaled_wealth = {
				to = FROM
				value = 2.0
				min = 200
				max = 800
			}
			prestige = 400
			add_character_modifier = {
				name = sotb_feud_victor
				duration = 5475
			}
		}
	}
	
	attacker_ai_victory_worth = {
		factor = -1 # Always accept unconditional surrender
	}
	
	attacker_ai_defeat_worth = {
		factor = 100
	}
	
	defender_ai_victory_worth = {
		factor = -1
	}
	
	defender_ai_defeat_worth = {
		factor = 100
	}
}
