# Event 50: Suspicion Slip-Up
character_event = {
	id = sotb_society.50
	title = EVTNAME_sotb_society_50
	desc = EVTDESC_sotb_society_50
	picture = GFX_evt_into_the_dungeon
	border = GFX_event_normal_frame_intrigue
	
	is_triggered_only = yes
	
	# Option A: Lie low and take the Suspected tag
	option = {
		name = EVTOPTA_sotb_society_50
		add_character_modifier = {
			name = sotb_suspected_conspirator
			duration = 1825
		}
	}
	
	# Option B: Total exposure!
	option = {
		name = EVTOPTB_sotb_society_50
		character_event = { id = sotb_society.55 }
	}
}

# Event 55: Unmasked as a Known Shadow Conspirator!
character_event = {
	id = sotb_society.55
	title = EVTNAME_sotb_society_55
	desc = EVTDESC_sotb_society_55
	picture = GFX_evt_into_the_dungeon
	border = GFX_event_normal_frame_war
	
	is_triggered_only = yes
	
	option = {
		name = EVTOPTA_sotb_society_55
		remove_character_modifier = sotb_suspected_conspirator
		add_character_modifier = {
			name = sotb_known_conspirator
			duration = -1
		}
		if = {
			limit = { independent = no }
			liege = {
				opinion = {
					who = ROOT
					modifier = opinion_known_shadow_conspirator
				}
			}
		}
	}
}
