# Event 100: Society Mission Issued (Infiltration & Spycraft)
character_event = {
	id = sotb_society.100
	title = EVTNAME_sotb_society_100
	desc = EVTDESC_sotb_society_100
	picture = GFX_evt_shadow
	border = GFX_event_normal_frame_intrigue
	
	trigger = {
		society_member_of = sotb_whispering_quill
		NOT = { has_character_flag = sotb_on_society_mission }
	}
	
	mean_time_to_happen = {
		months = 24
	}
	
	# Accept Mission
	option = {
		name = EVTOPTA_sotb_society_100
		set_character_flag = sotb_on_society_mission
		character_event = { id = sotb_society.101 days = 60 } # Mission outcome
	}
	
	# Decline Mission
	option = {
		name = EVTOPTB_sotb_society_100
		change_society_currency = -25
	}
}
