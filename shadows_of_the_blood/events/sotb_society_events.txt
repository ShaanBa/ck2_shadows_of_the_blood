# Event 50: The Shadow Slip-Up (Suspicion Raised)
character_event = {
	id = sotb_society.50
	title = EVTNAME_sotb_society_50
	desc = EVTDESC_sotb_society_50
	picture = GFX_evt_shadow
	border = GFX_event_normal_frame_intrigue
	
	is_triggered_only = yes
	
	# If not yet suspected -> becomes suspected
	option = {
		name = EVTOPTA_sotb_society_50
		trigger = {
			NOT = { has_character_modifier = sotb_suspected_conspirator }
			NOT = { has_character_modifier = sotb_known_conspirator }
		}
		add_character_modifier = {
			name = sotb_suspected_conspirator
			duration = 1825
		}
	}
	
	# If already suspected -> Full Public Unmasking!
	option = {
		name = EVTOPTB_sotb_society_50
		trigger = {
			has_character_modifier = sotb_suspected_conspirator
		}
		remove_character_modifier = sotb_suspected_conspirator
		character_event = { id = sotb_society.55 }
	}
}
