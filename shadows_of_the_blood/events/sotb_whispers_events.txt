# Event 1: The Investigation Concludes (30 Days After Dispatch)
character_event = {
	id = sotb_whispers.1
	title = EVTNAME_sotb_whispers_1
	desc = EVTDESC_sotb_whispers_1
	picture = GFX_evt_whispers
	border = GFX_event_normal_frame_intrigue
	
	is_triggered_only = yes
	
	immediate = {
		clr_character_flag = sotb_conducting_investigation
	}
	
	# Weighted Dynamic Outcome Resolution
	# 1. Critical Blackmail Goldmine (High Intrigue, Vulnerable Target)
	option = {
		name = EVTOPTA_sotb_whispers_1
		trigger = {
			OR = {
				intrigue = 12
				FROM = { trait = drunkard }
				FROM = { trait = lustful }
				FROM = { trait = trusting }
			}
		}
		character_event = { id = sotb_whispers.2 }
	}
	
	# 2. Minor Rumors & Intel (Moderate Success)
	option = {
		name = EVTOPTB_sotb_whispers_1
		character_event = { id = sotb_whispers.3 }
	}
	
	# 3. Dead End (Virtuous / Cautious Target)
	option = {
		name = EVTOPTC_sotb_whispers_1
		trigger = {
			OR = {
				FROM = { trait = honest }
				FROM = { trait = just }
				FROM = { trait = content }
				FROM = { intrigue = 14 }
			}
		}
		character_event = { id = sotb_whispers.4 }
	}
	
	# 4. Catastrophic Interception (Target's Spymaster catches and tortures your spy!)
	option = {
		name = EVTOPTD_sotb_whispers_1
		trigger = {
			FROM = {
				OR = {
					trait = paranoid
					trait = deceitful
					trait = elusive_shadow
					intrigue = 12
				}
			}
		}
		character_event = { id = sotb_whispers.5 }
	}
}
