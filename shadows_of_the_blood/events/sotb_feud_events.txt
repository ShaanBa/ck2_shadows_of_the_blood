# Event 10: The Challenge to Single Combat
character_event = {
	id = sotb_feud.10
	title = EVTNAME_sotb_feud_10
	desc = EVTDESC_sotb_feud_10
	picture = GFX_evt_duel
	border = GFX_event_normal_frame_war
	
	is_triggered_only = yes
	
	# Accept the Duel
	option = {
		name = EVTOPTA_sotb_feud_10
		trigger = {
			prisoner = no
			is_incapable = no
		}
		FROM = {
			character_event = { id = sotb_feud.11 } # Duel execution
		}
	}
	
	# Refuse the Duel (Cowardice)
	option = {
		name = EVTOPTB_sotb_feud_10
		prestige = -150
		add_trait = craven
		FROM = {
			prestige = 100
			opinion = {
				who = ROOT
				modifier = opinion_sotb_feud_slighted
			}
		}
	}
}
