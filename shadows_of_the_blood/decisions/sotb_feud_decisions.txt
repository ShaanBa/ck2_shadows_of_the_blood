# Shadows of the Blood - Feud Decisions

targetted_decisions = {
	
	# Formally escalate bad blood to an active Blood Feud
	sotb_declare_blood_feud = {
		filter = independent_rulers
		ai_target_filter = independent_rulers
		
		from_potential = {
			is_ruler = yes
			is_adult = yes
			prisoner = no
			NOT = { has_character_flag = sotb_active_feud }
		}
		
		potential = {
			is_ruler = yes
			is_adult = yes
			prisoner = no
			NOT = { dynasty = FROM }
			NOT = { has_character_flag = sotb_active_feud }
		}
		
		allow = {
			prestige = 200
			OR = {
				is_rival = FROM
				has_opinion_modifier = { who = FROM modifier = opinion_sotb_feud_slighted }
				opinion = { who = FROM value = -50 }
			}
		}
		
		effect = {
			FROM = {
				character_event = { id = sotb_feud.1 } # Event to launch the feud
			}
		}
		
		revoke_allowed = {
			always = no
		}
		
		ai_will_do = {
			factor = 1
			modifier = {
				factor = 2.0
				trait = wroth
			}
			modifier = {
				factor = 2.0
				trait = proud
			}
			modifier = {
				factor = 0
				trait = craven
			}
		}
	}
	
	# Challenge rival house head to a public Judicial Duel of Honor
	sotb_challenge_to_honor_duel = {
		filter = all
		ai_target_filter = all
		
		from_potential = {
			is_adult = yes
			prisoner = no
			is_incapable = no
			has_character_flag = sotb_active_feud
		}
		
		potential = {
			is_adult = yes
			prisoner = no
			is_incapable = no
			has_character_flag = sotb_active_feud
			has_opinion_modifier = { who = FROM modifier = opinion_sotb_blood_feud }
		}
		
		allow = {
			combat_rating = 10
			custom_tooltip = {
				text = sotb_duel_cooldown_tt
				NOT = { has_character_modifier = sotb_recently_dueled }
			}
		}
		
		effect = {
			FROM = {
				character_event = { id = sotb_feud.10 } # Challenge event
			}
		}
		
		ai_will_do = {
			factor = 1
			modifier = {
				factor = 2.5
				trait = brave
			}
			modifier = {
				factor = 0.1
				trait = craven
			}
		}
	}
	
	# Attempt to end the feud through a Blood Peace Marriage
	sotb_propose_blood_peace = {
		filter = independent_rulers
		ai_target_filter = independent_rulers
		
		from_potential = {
			is_ruler = yes
			has_character_flag = sotb_active_feud
		}
		
		potential = {
			is_ruler = yes
			has_character_flag = sotb_active_feud
			has_opinion_modifier = { who = FROM modifier = opinion_sotb_blood_feud }
		}
		
		allow = {
			wealth = 100
			prestige = 150
		}
		
		effect = {
			wealth = -100
			FROM = {
				character_event = { id = sotb_feud.20 } # Propose truce
			}
		}
		
		ai_will_do = {
			factor = 1
			modifier = {
				factor = 3.0
				trait = content
			}
			modifier = {
				factor = 0
				trait = wroth
			}
		}
	}
}
