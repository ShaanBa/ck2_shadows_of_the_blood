# Shadows of the Blood - Intrigue & Shadow Decisions

targetted_decisions = {
	
	# Dispatch an espionage network to investigate a target court
	sotb_snoop_for_secrets = {
		filter = all
		ai_target_filter = realm
		
		from_potential = {
			is_ruler = yes
			is_adult = yes
			prisoner = no
			is_incapable = no
			NOT = { has_character_flag = sotb_conducting_investigation }
		}
		
		potential = {
			is_ruler = yes
			is_adult = yes
			NOT = { character = FROM }
			NOT = { has_character_modifier = sotb_recently_investigated }
		}
		
		allow = {
			FROM = {
				wealth = 50
				scaled_wealth = 0.20
				intrigue = 8
				prisoner = no
				is_incapable = no
				custom_tooltip = {
					text = sotb_no_debt_tt
					NOT = { is_in_debt = yes }
				}
			}
		}
		
		effect = {
			FROM = {
				wealth = -50
				set_character_flag = sotb_conducting_investigation
			}
			add_character_modifier = {
				name = sotb_recently_investigated
				duration = 365
				hidden = yes
			}
			# Launches 30-day investigation timer
			FROM = {
				character_event = { id = sotb_whispers.1 days = 30 }
			}
		}
		
		ai_will_do = {
			factor = 1
			modifier = {
				factor = 2.5
				FROM = { trait = deceitful }
			}
			modifier = {
				factor = 0
				FROM = { trait = honest }
			}
		}
	}
	
	# Blackmail a specific ruler you hold leverage on
	sotb_blackmail_for_favors = {
		filter = all
		ai_target_filter = realm
		
		from_potential = {
			is_adult = yes
			prisoner = no
			is_incapable = no
		}
		
		potential = {
			is_adult = yes
			prisoner = no
			has_opinion_modifier = { who = FROM modifier = opinion_sotb_leverage_held }
		}
		
		allow = {
			always = yes
		}
		
		effect = {
			FROM = {
				character_event = { id = sotb_whispers.10 }
			}
		}
		
		ai_will_do = {
			factor = 1
		}
	}
}
